<script>
    // --- Navigation Logic ---
    function showSection(sectionId) {
        document.querySelectorAll('.content-section').forEach(section => {
            section.classList.remove('active');
        });
        document.querySelectorAll('.nav-button').forEach(button => {
            button.classList.remove('active');
        });
        document.getElementById(sectionId).classList.add('active');
        
        const activeBtn = Array.from(document.querySelectorAll('.nav-button')).find(btn => {
            const onclickVal = btn.getAttribute('onclick');
            return onclickVal && onclickVal.includes(`'${sectionId}'`);
        });
        if (activeBtn) activeBtn.classList.add('active');
        window.scrollTo({top: 0, behavior: 'smooth'});
    }

    // --- Utility Function for Text Input Validation ---
    function validateInputs(inputIds, feedbackId) {
        const feedbackEl = document.getElementById(feedbackId);
        let allCorrect = true;
        let correctCount = 0;

        inputIds.forEach(id => {
            const el = document.getElementById(id);
            if (!el) return;
            const userAnswer = el.value.trim().toLowerCase();
            const correctAnswer = el.dataset.answer.toLowerCase();

            el.classList.remove('correct-input', 'incorrect-input');

            if (userAnswer === correctAnswer) {
                el.classList.add('correct-input');
                correctCount++;
            } else {
                el.classList.add('incorrect-input');
                allCorrect = false;
            }
        });

        if (allCorrect) {
            feedbackEl.textContent = 'Excellent! All correct!';
            feedbackEl.style.color = '#10B981';
        } else {
            feedbackEl.textContent = `You got ${correctCount} out of ${inputIds.length} correct. Try again!`;
            feedbackEl.style.color = '#EF4444';
        }
    }

    // --- Intro: Numbers Game ---
    function checkNumbersGame() {
        validateInputs(['num-21', 'num-49', 'num-15', 'num-62', 'num-94', 'num-56'], 'numbers-feedback');
    }

    // --- Intro: Drag & Drop Floors ---
    let draggedFloorItem = null;

    function initFloorDragDrop() {
        document.querySelectorAll('#floor-bank .draggable-item').forEach(item => {
            item.addEventListener('dragstart', (e) => {
                draggedFloorItem = item;
                e.dataTransfer.setData('text/plain', item.dataset.matchId);
                item.classList.add('dragging');
            });
            item.addEventListener('dragend', () => {
                item.classList.remove('dragging');
                draggedFloorItem = null;
            });
        });

        document.querySelectorAll('.drop-zone').forEach(zone => {
            zone.addEventListener('dragover', (e) => {
                e.preventDefault(); 
                zone.classList.add('drag-over');
            });
            zone.addEventListener('dragleave', () => {
                zone.classList.remove('drag-over');
            });
            zone.addEventListener('drop', (e) => {
                e.preventDefault();
                zone.classList.remove('drag-over');

                if (draggedFloorItem && zone.children.length === 0) {
                    zone.appendChild(draggedFloorItem);
                }
            });
        });
    }

    function checkFloorMatch() {
        const zones = ['basement', 'ground', 'first', 'second', 'third', 'fourth'];
        let correctCount = 0;
        const feedbackEl = document.getElementById('floor-feedback');

        zones.forEach(z => {
            const zone = document.getElementById(`zone-${z}`);
            const item = zone.querySelector('.draggable-item');
            zone.classList.remove('incorrect-drop', 'correctly-placed');

            if (item) {
                if (item.dataset.matchId === z) {
                    zone.classList.add('correctly-placed');
                    item.draggable = false;
                    item.classList.add('cursor-not-allowed');
                    correctCount++;
                } else {
                    zone.classList.add('incorrect-drop');
                    setTimeout(() => zone.classList.remove('incorrect-drop'), 500);
                }
            } else {
                zone.classList.add('incorrect-drop');
                setTimeout(() => zone.classList.remove('incorrect-drop'), 500);
            }
        });

        if (correctCount === zones.length) {
            feedbackEl.textContent = 'Fantastic! You found where everyone lives!';
            feedbackEl.style.color = '#10B981';
        } else {
            feedbackEl.textContent = `You placed ${correctCount} items correctly. Keep trying!`;
            feedbackEl.style.color = '#EF4444';
        }
    }

    function resetFloorMatch() {
        const bank = document.getElementById('floor-bank');
        document.querySelectorAll('.draggable-item').forEach(item => {
            item.draggable = true;
            item.classList.remove('cursor-not-allowed');
            bank.appendChild(item);
        });
        document.querySelectorAll('.drop-zone').forEach(zone => {
            zone.classList.remove('correctly-placed', 'incorrect-drop');
        });
        document.getElementById('floor-feedback').textContent = '';
    }
